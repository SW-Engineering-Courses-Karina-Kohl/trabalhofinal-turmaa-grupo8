# buscar infos mais recente do GitHub (estado atual das branches remotas)
# baixa do servidor, mas não altera arquivos locais
git fetch origin 

# atualiza minha branch com alterações do servidor
git pull <mome_do_repos_remoto> <branch_remota_desejada>

# trocar para uma brach existente 
git checkout feature/dados 
git switch feature/dados

# add arquivos para o próximo commit 
git add .   # add todos da pasta que estou 
git add <caminho>/<Arquivo>   # add apenas um arquivo 

# criar um commit (ponto de salvamento)
git commit -m "Implementa testes do ConfigMapper"

# enviar alterações para GitHub
# -u vincula a branch local atual à branch remota
git push -u <mome_do_repos_remoto> <nome_da_branch> # qual branch daquele rep remoto
git push -u origin feature/brenda/dados

# se fez especificação acima, para próximos commits:
git push 

# estou em uma branch e quero incorporar nela tudo o que existe na main
